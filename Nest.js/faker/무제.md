# Faker 라이브러리 활용 가이드

## 개요

Faker는 테스트용 가짜 데이터를 생성하는 라이브러리입니다. 정적인 하드코딩된 테스트 데이터 대신 현실적이고 다양한 데이터를 생성하여 더 견고한 테스트를 작성할 수 있습니다.
## 설치

```bash

npm install --save-dev @faker-js/faker

```

### 시나리오별 팩토리 메서드
```typescript

export class UserFactory {

// 기본 사용자

static createUser(overrides?: Partial<User>): User {

return {

...this.defaultUser(),

...overrides,

} as User;

}

  

// 인증된 사용자

static createVerifiedUser(overrides?: Partial<User>): User {

return this.createUser({

certified: true,

certifiedAt: faker.date.recent({ days: 30 }),

status: UserStatus.ACTIVE,

...overrides,

});

}

  

// 신규 사용자

static createNewUser(overrides?: Partial<User>): User {

return this.createUser({

level: 1,

certified: false,

certifiedAt: null,

createdAt: faker.date.recent({ days: 1 }),

...overrides,

});

}

  

// 차단된 사용자

static createSuspendedUser(overrides?: Partial<User>): User {

return this.createUser({

status: UserStatus.SUSPENDED,

certified: false,

refreshToken: null,

...overrides,

});

}

}
```

## Controller 테스트에서 Request DTO 활용
### Request DTO Factory 생성

  

```typescript

// test/factories/create-customer-request.factory.ts

import { faker } from '@faker-js/faker';

import { CreateCustomerRequest } from '@/user/dtos/create-customer.request';

  

export class CreateCustomerRequestFactory {

private static defaultRequest(): Partial<CreateCustomerRequest> {

return {

email: faker.internet.email(),

loginId: faker.internet.username(),

password: faker.internet.password({ length: 12 }),

name: faker.person.fullName(),

phone: faker.helpers.replaceSymbols('010-####-####'),

bizRegNumber: faker.helpers.replaceSymbols('###-##-#####'),

companyName: faker.company.name(),

address: faker.location.streetAddress(),

detailAddress: faker.location.secondaryAddress(),

};

}

  

static create(overrides?: Partial<CreateCustomerRequest>): CreateCustomerRequest {

return {

...this.defaultRequest(),

...overrides,

} as CreateCustomerRequest;

}

  

// 유효한 요청 데이터

static createValid(): CreateCustomerRequest {

return this.create({

password: 'ValidPassword123!',

bizRegNumber: '123-45-67890',

});

}

  

// 잘못된 요청 데이터

static createInvalid(): CreateCustomerRequest {

return this.create({

email: 'invalid-email',

password: '123',

phone: '123',

});

}

  

// 중복 데이터

static createDuplicate(): CreateCustomerRequest {

return this.create({

loginId: 'existing-user',

bizRegNumber: '999-99-99999',

});

}

  

// 사업자 타입별 생성

static createWithBizType(bizType: 'INDIVIDUAL' | 'CORPORATION'): CreateCustomerRequest {

const base = this.createValid();

if (bizType === 'INDIVIDUAL') {

return {

...base,

bizRegNumber: faker.helpers.replaceSymbols('###-##-#####'),

companyName: null,

};

} else {

return {

...base,

bizRegNumber: faker.helpers.replaceSymbols('###-##-#####'),

companyName: faker.company.name(),

};

}

}

  

// 대량 생성

static createBatch(count: number, overrides?: Partial<CreateCustomerRequest>): CreateCustomerRequest[] {

return Array.from({ length: count }, () => this.create(overrides));

}

}

```

### Controller 테스트에서 활용

```typescript

// test/user/user.controller.spec.ts

import { CreateCustomerRequestFactory } from '../factories/create-customer-request.factory';

  

describe('UserController', () => {

let controller: UserController;

let userService: jest.Mocked<IUserService>;

  

describe('POST /users/customers', () => {

it('성공적으로 고객을 생성해야 한다', async () => {

// Arrange

const request = CreateCustomerRequestFactory.createValid();

userService.createCustomer.mockResolvedValue(undefined);

  

// Act

await controller.createCustomer(request);

  

// Assert

expect(userService.createCustomer).toHaveBeenCalledWith(request);

});

  

it('잘못된 요청 시 400 에러를 반환해야 한다', async () => {

// Arrange

const invalidRequest = CreateCustomerRequestFactory.createInvalid();

  

// Act & Assert

await expect(controller.createCustomer(invalidRequest))

.rejects.toThrow(BadRequestException);

});

  

it('중복된 데이터 시 409 에러를 반환해야 한다', async () => {

// Arrange

const duplicateRequest = CreateCustomerRequestFactory.createDuplicate();

userService.createCustomer.mockRejectedValue(

new UserException(사용자예외목록.중복된_로그인_아이디)

);

  

// Act & Assert

await expect(controller.createCustomer(duplicateRequest))

.rejects.toThrow(ConflictException);

});

});

});

```

  

### E2E 테스트에서 활용

  

```typescript

// test/user/user.e2e-spec.ts

describe('User E2E', () => {

it('/users/customers (POST)', () => {

const request = CreateCustomerRequestFactory.createValid();

  

return request(app.getHttpServer())

.post('/users/customers')

.send(request)

.expect(201);

});

  

it('여러 사용자 생성 테스트', async () => {

const requests = Array.from({ length: 10 }, () =>

CreateCustomerRequestFactory.createValid()

);

  

for (const req of requests) {

await request(app.getHttpServer())

.post('/users/customers')

.send(req)

.expect(201);

}

});

  

it('대량 사용자 처리 테스트', async () => {

const requests = CreateCustomerRequestFactory.createBatch(100);

const promises = requests.map(req =>

request(app.getHttpServer())

.post('/users/customers')

.send(req)

.expect(201)

);

  

await Promise.all(promises);

});

});

```

```typescript
faker.number.float({ min: 0, max: 1 }) // 소수

  

// 불린

faker.datatype.boolean(0.8) // 80% 확률로 true

  

// 날짜

faker.date.recent({ days: 7 }) // 최근 날짜

faker.date.between({ from: startDate, to: endDate })
```

## Faker 활용의 장점

  

### 1. 현실적인 테스트 데이터

- 매번 다른 데이터로 더 많은 케이스 테스트

- 실제 운영 환경과 유사한 데이터

  

### 2. 다양성과 견고함

- 하드코딩된 데이터에서 놓칠 수 있는 edge case 발견

- 데이터 변화에 따른 버그 사전 발견

  

### 3. 유지보수성

- Factory 패턴으로 중앙화된 테스트 데이터 관리

- 시나리오별 데이터 생성으로 테스트 의도 명확화

  

### 4. 확장성

- 대량 데이터 생성으로 성능 테스트 지원

- 새로운 시나리오 추가 용이

  

## 주의사항

  

### 1. 결정론적 테스트가 필요한 경우

```typescript

// 시드 설정으로 재현 가능한 데이터 생성

faker.seed(12345);

const data = faker.person.fullName(); // 항상 동일한 결과

```

  

### 2. 특정 형식이 중요한 경우

```typescript

// 한국 전화번호 형식

faker.helpers.replaceSymbols('010-####-####')

  

// 한국 사업자번호 형식

faker.helpers.replaceSymbols('###-##-#####')

```

  

### 3. 관련 데이터 일관성

```typescript

// 생성일이 수정일보다 나중이 되는 문제 방지

const updatedAt = faker.date.between({ from: createdAt, to: new Date() });

```
